cmake_minimum_required(VERSION 3.16)
project(RPMS_CPP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG v0.18.3 GIT_SHALLOW TRUE)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(httplib json)
add_library(rpms_core src/database.cpp)
target_include_directories(rpms_core PUBLIC include)
target_link_libraries(rpms_core PUBLIC SQLite::SQLite3 OpenSSL::Crypto)
target_compile_options(rpms_core PRIVATE
  $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic -Wconversion>
  $<$<CXX_COMPILER_ID:MSVC>:/W4>)
add_executable(rpms_create_user tools/create_user.cpp)
target_link_libraries(rpms_create_user PRIVATE rpms_core)
add_executable(rpms_server src/main.cpp)
target_link_libraries(rpms_server PRIVATE rpms_core httplib::httplib nlohmann_json::nlohmann_json Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(rpms_database_tests tests/database_tests.cpp)
  target_link_libraries(rpms_database_tests PRIVATE rpms_core)
  add_executable(rpms_api_test_seed tests/api_test_seed.cpp)
  target_link_libraries(rpms_api_test_seed PRIVATE rpms_core)
  add_test(NAME rpms_database_tests COMMAND rpms_database_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  find_program(NODE_EXECUTABLE node)
  if(NODE_EXECUTABLE)
    add_test(NAME rpms_frontend_contract_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/frontend_contract_test.py ${NODE_EXECUTABLE})
  else()
    add_test(NAME rpms_frontend_contract_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/frontend_contract_test.py)
  endif()
  add_test(NAME rpms_http_api_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/api_smoke_test.py $<TARGET_FILE:rpms_server> $<TARGET_FILE:rpms_api_test_seed>)
  set_tests_properties(rpms_http_api_tests PROPERTIES TIMEOUT 60)
endif()
